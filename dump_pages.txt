# Kinds that are always worth surfacing on the Dashboard even amid a run of
# routine scans - anything NOT in this "routine" set is always shown.
# scan_log.py/activity_log.py themselves are untouched by this - it only
# changes what the Dashboard's own Recent Activity list chooses to surface.
_ROUTINE_ACTIVITY_KINDS = {"scan_completed"}
_ROUTINE_DEDUP_WINDOW_SECONDS = 10 * 60


def _filter_meaningful_events(events):
    """
    events: activity_log entries, newest first (as recent_events() returns).
    Collapses back-to-back routine "scan completed, nothing found" entries
    within a short window down to just the newest one, so five near-
    identical "Quick Scan completed" rows don't crowd out anything else
    that happened. A routine scan that DID find something (its own detail
    text says so - see AppState.finalize_scan()'s summary) is never
    collapsed - only genuinely uneventful completions are.
    """
    filtered = []
    last_routine_at = None
    for event in events:
        is_routine = (
            event.get("kind") in _ROUTINE_ACTIVITY_KINDS
            and "No threats found" in (event.get("detail") or "")
        )
        if is_routine:
            try:
                this_time = datetime.fromisoformat(event["timestamp"])
            except (KeyError, ValueError):
                this_time = None
            if last_routine_at is not None and this_time is not None:
                if abs((last_routine_at - this_time).total_seconds()) < _ROUTINE_DEDUP_WINDOW_SECONDS:
                    continue  # near-duplicate of the routine completion we already kept
            if this_time is not None:
                last_routine_at = this_time
        filtered.append(event)
    return filtered

class QuickActionCard(QWidget):
    clicked = Signal()

    # Exact reference values: the primary (Quick Scan) card's icon is
    # #B9FFDA and its label is #F2FFF7; the other three cards' icon is
    # #7AA98C and their label is #C7E3D3 - two genuinely distinct tiers,
    # not one shared default recolored on hover.
    _ICON_HOVER_COLOR = theme.CYBER_NEON

    def __init__(self, icon_name, label, primary=False, disabled=False, caption=None, parent=None):
        super().__init__(parent)
        # A plain QWidget subclass does NOT automatically track mouse-enter/
        # leave for :hover QSS the way QAbstractButton/QFrame-derived
        # widgets do - confirmed live: the #quickActionCard:hover rule was
        # syntactically fine but never actually repainted the card without
        # this. Needed for the hover CSS (and the icon-hover recolor below)
        # to have any visual effect at all. Set BEFORE apply_glow() below
        # (not after) - theme.apply_glow() checks WA_StyledBackground to
        # decide whether it's safe to also set WA_TranslucentBackground
        # (the seam-artifact fix): the two together break this widget's own
        # QSS background/border painting entirely, confirmed live via a
        # rendered comparison grid (the Quick Scan card silently lost its
        # background/border, just the icon/text floating with nothing
        # behind it) - order matters here, this must run first.
        self.setAttribute(Qt.WA_Hover, True)
        self.setAttribute(Qt.WA_StyledBackground, True)
        if disabled:
            self.setObjectName("quickActionCardDisabled")
        elif primary:
            self.setObjectName("quickActionCardPrimary")
            # box-shadow 0 0 18px rgba(60,220,140,0.25) - Quick Scan card's
            # exact glow.
            theme.apply_glow(self, theme.CYBER_GLOW_NEON, blur=18)
        else:
            self.setObjectName("quickActionCard")
        self.setCursor(Qt.PointingHandCursor if not disabled else Qt.ArrowCursor)
        self.setFixedHeight(100)
        self._disabled = disabled
        self._primary = primary
        self._icon_default_color = theme.PRIMARY_SCAN_ICON if primary else theme.CYBER_TEXT_SECONDARY

        layout = QVBoxLayout(self)
        layout.setAlignment(Qt.AlignCenter)
        layout.setSpacing(6)

        icon_color = theme.CYBER_TEXT_MUTE if disabled else self._icon_default_color
        self._icon_widget = icon_label(icon_name, size=20, color=icon_color)
        layout.addWidget(self._icon_widget, alignment=Qt.AlignHCenter)

        text_color = theme.CYBER_TEXT_MUTE if disabled else (theme.CYBER_TEXT_PRIMARY if primary else theme.CYBER_TEXT_SECONDARY)
        text = make_label(label, "bodyText")
        text.setStyleSheet(f"font-size: 11px; font-weight: 500; color: {text_color};")
        text.setAlignment(Qt.AlignCenter)
        layout.addWidget(text)

        if caption:
            cap = make_label(caption, "metadataDisabled")
            cap.setAlignment(Qt.AlignCenter)
            layout.addWidget(cap)

    def mousePressEvent(self, event):
        if not self._disabled:
            self.clicked.emit()
        super().mousePressEvent(event)

    def enterEvent(self, event):
        if not self._disabled:
            self._icon_widget.setStyleSheet(icon_style(self._ICON_HOVER_COLOR))
            theme.apply_glow(self._icon_widget, theme.CYBER_NEON, blur=10, alpha=150)
        super().enterEvent(event)

    def leaveEvent(self, event):
        if not self._disabled:
            self._icon_widget.setStyleSheet(icon_style(self._icon_default_color))
            theme.clear_glow(self._icon_widget)
        super().leaveEvent(event)

    def _hero_accent_dot(self):
        """
        The single, deliberately tiny decorative warm-amber touch called for
        in the new visual language - a small highlight point on the hero
        card only, never anywhere functional/interactive. It carries no
        status meaning (it's identical whether protection is on or off) and
        is never clicked - purely a nod to the reference image's one warm
        accent amid all the green.
        """
        dot = QLabel()
        dot.setFixedSize(6, 6)
        dot.setStyleSheet(f"background-color: {theme.BRAND_ORANGE}; border-radius: 3px;")
        theme.apply_glow(dot, theme.BRAND_ORANGE, blur=10, alpha=120)
        return dot

    def _clear_layout(self, layout):
        while layout.count():
            item = layout.takeAt(0)
            widget = item.widget()
            if widget is not None:
                widget.deleteLater()
            elif item.layout() is not None:
                self._clear_layout(item.layout())

            self.hero_card.addWidget(subtext)

            self.hero_card.addWidget(hairline())
            last_scan_raw = self.app_state.last_scan_at()
            last_scan = format_timestamp(last_scan_raw) if last_scan_raw else "Never"

            # (not yet quarantined/deleted/ignored), so "Open Threats" is

            self.hero_card.addWidget(stat_block([("LAST SCAN", last_scan), ("OPEN THREATS", open_threats)]))

            subtext.setStyleSheet(f"font-size: 12px; color: {theme.CYBER_TEXT_SECONDARY};")
            self.hero_card.addWidget(subtext)

            btn.setObjectName("primaryButton")

            btn.clicked.connect(lambda: self.app_state.set_realtime_protection(True))

    def _refresh_quick_actions(self):
        while self._quick_actions_row.count():
            item = self._quick_actions_row.takeAt(0)
            if item.widget():
                item.widget().deleteLater()

        drives = self.app_state.connected_removable_drives()
        for icon_name, label, primary, key in _QUICK_ACTIONS:
            disabled = key == "usb" and len(drives) == 0
            caption = "No drive" if (key == "usb" and disabled) else None
            card = QuickActionCard(icon_name, label, primary=primary, disabled=disabled, caption=caption)
            card.clicked.connect(lambda k=key: self.scan_requested.emit(k))
            self._quick_actions_row.addWidget(card)

    def _refresh_overview(self):
        self._clear_layout(self._overview_body)
        protected = self.app_state.is_realtime_protection_enabled()

        # Real-Time Protection: the main item.
        self._overview_body.addWidget(
            status_row("Real-Time Protection", "active" if protected else "inactive",
                       "Active" if protected else "Off")
        )

        # Behavior Monitoring / Downloads Protection / USB Auto-Scan only
        # ever run because Real-Time Protection is on, so they're grouped
        # as indented sub-items beneath it (smaller text, a subtle left
        # rule) rather than three more independent-looking top-level rows.
        sub_items = [
            ("Behavior Monitoring", protected and self.app_state.is_module_enabled("behavior_monitoring")),
            ("Downloads Protection", protected and self.app_state.is_module_enabled("downloads_autoscan")),
            ("USB Auto-Scan", protected and self.app_state.is_module_enabled("usb_autoscan")),
        ]
        sub_group = QFrame()
        sub_group.setObjectName("overviewSubgroup")
        sub_layout = QVBoxLayout(sub_group)
        sub_layout.setContentsMargins(16, 2, 0, 2)
        sub_layout.setSpacing(0)
        for label, active in sub_items:
            sub_layout.addWidget(
                status_row(label, "active" if active else "inactive", "Active" if active else "Off", compact=True)
            )
        self._overview_body.addWidget(sub_group)
        self._overview_body.addWidget(hairline())

        # Windows Startup: unrelated to real-time protection - stays a
        # separate top-level item, not part of the group above.
        startup = self.app_state.is_startup_enabled()
        self._overview_body.addWidget(
            status_row("Windows Startup", "active" if startup else "inactive", "Enabled" if startup else "Off")
        )

    def _refresh_activity(self):
        self._clear_layout(self._activity_body)
        pool = activity_log.recent_events(limit=30)
        meaningful = _filter_meaningful_events(pool)
        events = meaningful[:5]

        if not events:
            empty = EmptyState("You're all clear", "No recent security issues have been detected.")
            empty.setFixedHeight(140)
            self._activity_body.addWidget(empty)
            self._view_all_btn.hide()
            return

        for event in events:
            row = QHBoxLayout()
            row.setSpacing(10)
            attention_kinds = {"threat_detected", "behavior_alert"}
            dot = StatusDot("attention" if event["kind"] in attention_kinds else "active")
            row.addWidget(dot)
            col = QVBoxLayout()
            col.setSpacing(2)
            col.addWidget(body_text(event["title"]))
            when = format_timestamp(event["timestamp"])
            col.addWidget(make_label(when, "metadataText"))
            row.addLayout(col)
            row.addStretch()
            self._activity_body.addLayout(row)

        self._view_all_btn.setVisible(len(meaningful) > 5)

def _format_elapsed(seconds):
    seconds = int(seconds)
    minutes, secs = divmod(seconds, 60)
    return f"{minutes:02d}:{secs:02d}"


def _elide_left(text, max_len=64):
    if text is None:
        return "-"
    if len(text) <= max_len:
        return text
    return "..." + text[-(max_len - 3):]

    def _build_job_from_tree(self):
        checked = set(self.tree.checked_ids())
        drive_items = self.app_state.list_fixed_drive_items()
        drive_paths = [d["path"] for d in drive_items if d["id"] in checked]

        location_map = self.app_state.common_location_paths()
        location_paths = [path for item_id, path in location_map.items() if item_id in checked]

        system_area = bool(checked & set(SYSTEM_AREA_IDS)) or "sys" in checked
        deep_inspection = bool(checked & set(DEEP_INSPECTION_IDS)) or "tools" in checked

        usb_path = None
        if "usb" in checked:
            drives = self.app_state.connected_removable_drives()
            if drives:

        return self.app_state.custom_scan_job(drive_paths, location_paths, system_area, deep_inspection, usb_path)

        if persistence_entries:

"""
scan_tree.py

The Scan page's mode-radio-row + checkbox tree (assets/scan_page_reference.html,
confirmed with the user - see that file's inline <script> for the exact
selection/expand-collapse behavior this reproduces in PySide6). Computer
Drive's children are the only dynamic part - populated from the real fixed
drives on this machine (see AppState.list_fixed_drive_items()), everything
else is the reference's fixed tree shape.

Behavior (kept identical to the reference's plain-JS implementation):
  - Clicking a mode radio re-applies that mode's default checked set,
    replacing whatever was checked before.
  - Clicking a GROUP checkbox toggles all its children to match it.
  - Clicking a CHILD checkbox toggles only that child - it never touches
    its parent's or siblings' checked state.
  - Clicking a group's arrow toggles that group's expanded/collapsed
    state only - it never touches any checked state.
  - Which mode radio reads as "selected" only changes when a radio itself
    is clicked - manually editing checkboxes afterward does not clear it
    (exactly the reference's `current` variable semantics).
"""

class ScanTreeWidget(QWidget):
    """
    The checkbox tree box only (no mode radios - see ScanModeRow, a
    separate sibling widget the Scan page wires to apply_mode() below, so
    the page can lay the two out exactly like the reference: the mode row
    full-width above, the tree box and the Scan button side by side below
    it). `tree_spec` is built once (via build_tree_spec()) and passed in -
    this widget owns no knowledge of where drive items come from, only how
    to render/toggle a flat list of {id, text, indent, parent, group?} rows.
    """

    selection_changed = Signal()

    def __init__(self, tree_spec, parent=None):
        super().__init__(parent)
        self.tree_spec = tree_spec
        self.mode_checks = default_mode_checks(tree_spec)
        self.children_of = {}
        for item in tree_spec:
            if item["parent"] is not None:
                self.children_of.setdefault(item["parent"], []).append(item["id"])

        self.state = {item["id"]: False for item in tree_spec}
        self.expanded = {item["id"]: True for item in tree_spec if item.get("group")}
        self.current_mode = None

        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.setSpacing(0)

        self.tree_scroll = QScrollArea()
        self.tree_scroll.setWidgetResizable(True)
        self.tree_scroll.setObjectName("scanTreeBox")
        self.tree_scroll.setFixedHeight(260)
        tree_body = QWidget()
        tree_body.setObjectName("scanTreeBody")
        self.tree_layout = QVBoxLayout(tree_body)
        self.tree_layout.setContentsMargins(14, 12, 14, 12)
        self.tree_layout.setSpacing(2)
        self.tree_scroll.setWidget(tree_body)
        outer.addWidget(self.tree_scroll)

        self.tree_scroll.setStyleSheet(f"""
            #scanTreeBox {{
                border-radius: {theme.RADIUS_CARD}px;
                border: 1px solid {theme.CYBER_BORDER_DEFAULT};
                background: {theme.GRADIENT_NEUTRAL_CARD};
            }}
            #scanTreeBody {{ background: transparent; }}
        """)

        self._rows = {}
        for item in tree_spec:
            row = _TreeRow(item, self._on_checkbox_toggled, self._on_arrow_clicked)
            self._rows[item["id"]] = row
            self.tree_layout.addWidget(row)
        self.tree_layout.addStretch()

        self.apply_mode("quick")

    # ------------------------------------------------------------------
    def apply_mode(self, mode):
        self.current_mode = mode
        checked_ids = set(self.mode_checks[mode])
        for item_id in self.state:
            self.state[item_id] = item_id in checked_ids
        self._render_rows()
        self.selection_changed.emit()

    def checked_ids(self):
        return [item_id for item_id, checked in self.state.items() if checked]

    def is_checked(self, item_id):
        return bool(self.state.get(item_id))

    def any_checked(self, item_ids):
        return any(self.state.get(i) for i in item_ids)

    # ------------------------------------------------------------------
    def _on_checkbox_toggled(self, item_id, checked):
        if self.state.get(item_id) == checked:
            return
        self.state[item_id] = checked
        if item_id in self.children_of:  # a group checkbox - cascade to children
            for child_id in self.children_of[item_id]:
                self.state[child_id] = checked
        self._render_rows()
        self.selection_changed.emit()

    def _on_arrow_clicked(self, group_id):
        self.expanded[group_id] = not self.expanded[group_id]
        self._render_rows()

    def _render_rows(self):
        for item in self.tree_spec:
            row = self._rows[item["id"]]
            parent = item["parent"]
            visible = parent is None or self.expanded.get(parent, True)
            row.setVisible(visible)
            row.set_checked_silently(self.state[item["id"]])
            if item.get("group"):
                row.set_arrow_expanded(self.expanded.get(item["id"], True))

    def _build_exceptions(self):
        widget, layout = self._page_container()
        card = Card()
        card.addWidget(section_title("Excluded locations"))
        self.exceptions_list_layout = QVBoxLayout()
        card.addLayout(self.exceptions_list_layout)

        add_row = QHBoxLayout()
        add_btn = QPushButton("+ Add Exception")
        add_btn.setObjectName("secondaryButton")
        add_btn.setCursor(Qt.PointingHandCursor)
        add_btn.clicked.connect(self._add_exception)
        add_row.addWidget(add_btn)
        add_row.addStretch()
        card.addLayout(add_row)

        layout.addWidget(card)
        layout.addStretch()
        self._refresh_exceptions()
        return widget

    def _refresh_exceptions(self):
        while self.exceptions_list_layout.count():
            item = self.exceptions_list_layout.takeAt(0)
            if item.widget():
                item.widget().deleteLater()

        paths = self.app_state.list_exceptions()
        if not paths:
            note = make_label(
                "No exceptions added - files and folders you exclude from scanning will appear here.",
                "metadataDisabled",
            )
            note.setWordWrap(True)
            self.exceptions_list_layout.addWidget(note)
            return

        for path in paths:
            row = QHBoxLayout()
            label = body_text(path, wrap=True)
            row.addWidget(label, 1)
            remove_btn = QPushButton("Remove")
            remove_btn.setObjectName("textButton")
            remove_btn.setCursor(Qt.PointingHandCursor)
            remove_btn.clicked.connect(lambda _checked=False, p=path: self._remove_exception(p))
            row.addWidget(remove_btn)
            row_widget = QWidget()
            row_widget.setObjectName("transparentPanel")
            row_widget.setLayout(row)
            self.exceptions_list_layout.addWidget(row_widget)
            self.exceptions_list_layout.addWidget(hairline())

    def _add_exception(self):
        path = QFileDialog.getExistingDirectory(self, "Add Folder Exception")
        if not path:
            path, _ = QFileDialog.getOpenFileName(self, "Add File Exception")
        if path:
            self.app_state.add_exception_path(path)
            self._refresh_exceptions()

    def _remove_exception(self, path):
        if ConfirmationDialog.confirm(
            self, "Remove this exception?",
            f"DEVADARA will resume scanning '{path}'.",
            confirm_label="Remove",
        ):
            self.app_state.remove_exception_path(path)
            self._refresh_exceptions()

    def _build_notifications(self):
        widget, layout = self._page_container()
        card = Card()
        specs = [
            ("Scan completed", "notify_scan_completed"),
            ("Threat detected", "notify_threat_detected"),
            ("USB scan completed", "notify_usb_scan_completed"),
            ("Protection status changes", "notify_protection_status"),

        for index, (label, key) in enumerate(specs):
            row, switch = toggle_row(label)

        layout.addStretch()
        return widget
